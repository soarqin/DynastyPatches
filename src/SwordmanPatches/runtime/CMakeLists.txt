cmake_minimum_required(VERSION 3.20)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/../../../cmake")
include(CustomCompilerOptions)
include(CastlePatchesVersion)
project(SwordmanRuntime VERSION ${CASTLEPATCHES_VERSION} LANGUAGES C)
include(Dependencies)
set(CMAKE_ASM_NASM_OBJECT_FORMAT win32)
enable_language(ASM_NASM)
add_library(SwordmanRuntime SHARED
    runtime_dll.c ../file_identity.c gameplay.c gameplay_hooks.asm gameplay_hook_addresses.inc
    display.c display_hooks.asm
    SwordmanRuntime.def "${CMAKE_CURRENT_BINARY_DIR}/SwordmanRuntimeVersion.rc")
set(SWORDMANPATCHES_FILE_TYPE VFT_DLL)
set(SWORDMANPATCHES_FILE_DESCRIPTION "Swordman Patches Runtime")
set(SWORDMANPATCHES_INTERNAL_NAME "SwordmanRuntime")
set(SWORDMANPATCHES_ORIGINAL_FILENAME "SwordmanRuntime.dll")
configure_file(../SwordmanPatchesVersion.rc.in SwordmanRuntimeVersion.rc @ONLY)
target_compile_definitions(SwordmanRuntime PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(SwordmanRuntime PRIVATE . ..)
target_link_libraries(SwordmanRuntime PRIVATE dbghelp minhook bcrypt gdi32 user32)
if(MSVC)
    target_compile_options(SwordmanRuntime PRIVATE $<$<COMPILE_LANGUAGE:C>:/W4 /permissive- /utf-8>)
else()
    target_compile_options(SwordmanRuntime PRIVATE -Wall -Wextra -Wpedantic)
endif()
